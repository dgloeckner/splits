# FEAT-0012 Race Boat Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a 3-lane "race boat" tile to the live dashboard that pits the current row against a ghost (last matching session) and a PB-leader (workout-PB or matching distance-PB), with off-screen gap chips and an end-of-session toast.

**Architecture:** A new backend endpoint `GET /api/v1/race/opponents?workout_id=...&rower_id=...` resolves both opponents at session start by reusing the existing `workout.shape_hash` and `pb_workout` / `pb_distance` tables, returning each opponent's `t_sec` and `split` arrays from `summary_json.samples`. The frontend fetches opponents once at session start, reconstructs `distance(t)` client-side via cumulative pace integration with monotone-cubic interpolation, renders a center-on-user side-scroll lane in a new `race-boat` tile type, and emits an ephemeral toast at session end. No new persistence — race results are toast-only.

**Tech Stack:** Go (`net/http` + `database/sql`), React + TypeScript + Vite, vanilla CSS modules. Existing helpers: `backend/internal/workout.ShapeHash`, `backend/internal/stats/samples`, `web/src/live/smoothing.ts` (monotone-cubic), `web/src/tiles/registry.ts`, `web/src/icons/IconBase.tsx`.

```go
const ghostQuery = `
SELECT s.id, s.summary_json
FROM sessions s
JOIN workout w ON w.id = s.workout_id
WHERE s.rower_id = ?
  AND w.shape_hash = (SELECT shape_hash FROM workout WHERE id = ?)
  AND s.ended_at IS NOT NULL
  AND s.deleted_at IS NULL
ORDER BY s.started_at DESC
LIMIT 1`

func RaceOpponentsHandler(db *sql.DB) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		workoutID := r.URL.Query().Get("workout_id")
		rowerID := r.URL.Query().Get("rower_id")
		if workoutID == "" || rowerID == "" {
			http.Error(w, "workout_id and rower_id required", http.StatusBadRequest)
			return
		}
		resp := RaceOpponentsResponse{}
		if g, err := loadGhost(r.Context(), db, rowerID, workoutID); err != nil {
			http.Error(w, "ghost lookup: "+err.Error(), http.StatusInternalServerError)
			return
		} else {
			resp.Ghost = g
		}
		w.Header().Set("Content-Type", "application/json")
		_ = json.NewEncoder(w).Encode(resp)
	})
}

type samplesPayload struct {
	Samples struct {
		TSec  []float64 `json:"t_sec"`
		Split []float64 `json:"split"`
	} `json:"samples"`
}

func loadGhost(ctx context.Context, db *sql.DB, rowerID, workoutID string) (*RaceOpponent, error) {
	var sid, summary string
	err := db.QueryRowContext(ctx, ghostQuery, rowerID, workoutID).Scan(&sid, &summary)
	if err == sql.ErrNoRows {
		return nil, nil
	}
	if err != nil {
		return nil, err
	}
	var sp samplesPayload
	if err := json.Unmarshal([]byte(summary), &sp); err != nil {
		return nil, err
	}
	if len(sp.Samples.TSec) == 0 {
		return nil, nil
	}
	return &RaceOpponent{
		SourceSessionID: sid, Label: "ghost",
		TSec: sp.Samples.TSec, Split: sp.Samples.Split,
	}, nil
}
```

```go
const pbWorkoutQuery = `
SELECT s.id, s.summary_json, pw.elapsed_sec
FROM pb_workout pw
JOIN sessions s ON s.id = pw.source_session_id
WHERE pw.rower_id = ?
  AND pw.shape_hash = (SELECT shape_hash FROM workout WHERE id = ?)
  AND pw.invalidated_at IS NULL
  AND s.deleted_at IS NULL
LIMIT 1`

const pbDistanceQuery = `
SELECT s.id, s.summary_json, pd.elapsed_sec
FROM pb_distance pd
JOIN sessions s ON s.id = pd.source_session_id
WHERE pd.rower_id = ?
  AND pd.distance_m = ?
  AND pd.invalidated_at IS NULL
  AND s.deleted_at IS NULL
LIMIT 1`

func loadPB(ctx context.Context, db *sql.DB, rowerID, workoutID string) (*RaceOpponent, error) {
	wPB, wPBSec, wSummary, ok, err := queryPBWorkout(ctx, db, rowerID, workoutID)
	if err != nil {
		return nil, err
	}
	dist, hasDist := standardDistanceForWorkout(ctx, db, workoutID)
	var dPB string
	var dPBSec float64
	var dSummary string
	hasDistPB := false
	if hasDist {
		dPB, dSummary, dPBSec, hasDistPB, err = queryPBDistance(ctx, db, rowerID, dist)
		if err != nil {
			return nil, err
		}
	}
	switch {
	case hasDistPB && (!ok || dPBSec < wPBSec):
		return opponentFromSummary(dPB, "pb", dSummary)
	case ok:
		return opponentFromSummary(wPB, "pb", wSummary)
	}
	return nil, nil
}

func queryPBWorkout(ctx context.Context, db *sql.DB, rowerID, workoutID string) (id string, elapsed float64, summary string, ok bool, err error) {
	row := db.QueryRowContext(ctx, pbWorkoutQuery, rowerID, workoutID)
	err = row.Scan(&id, &summary, &elapsed)
	if err == sql.ErrNoRows {
		return "", 0, "", false, nil
	}
	if err != nil {
		return "", 0, "", false, err
	}
	return id, elapsed, summary, true, nil
}

func queryPBDistance(ctx context.Context, db *sql.DB, rowerID string, distance int) (id string, summary string, elapsed float64, ok bool, err error) {
	row := db.QueryRowContext(ctx, pbDistanceQuery, rowerID, distance)
	err = row.Scan(&id, &summary, &elapsed)
	if err == sql.ErrNoRows {
		return "", "", 0, false, nil
	}
	if err != nil {
		return "", "", 0, false, err
	}
	return id, summary, elapsed, true, nil
}

// standardDistanceForWorkout returns the int meters for a fixed-distance
// workout whose params match a known PB-distance bucket; returns
// (0, false) otherwise.
func standardDistanceForWorkout(ctx context.Context, db *sql.DB, workoutID string) (int, bool) {
	var mode, params string
	if err := db.QueryRowContext(ctx,
		`SELECT mode, params FROM workout WHERE id = ?`, workoutID).
		Scan(&mode, &params); err != nil {
		return 0, false
	}
	if mode != "fixed-distance" {
		return 0, false
	}
	var p struct {
		Meters int `json:"meters"`
	}
	if err := json.Unmarshal([]byte(params), &p); err != nil {
		return 0, false
	}
	switch p.Meters {
	case 500, 1000, 2000, 5000, 6000, 10000, 21097, 42195:
		return p.Meters, true
	}
	return 0, false
}

func opponentFromSummary(sourceID, label, summary string) (*RaceOpponent, error) {
	var sp samplesPayload
	if err := json.Unmarshal([]byte(summary), &sp); err != nil {
		return nil, err
	}
	if len(sp.Samples.TSec) == 0 {
		return nil, nil
	}
	return &RaceOpponent{
		SourceSessionID: sourceID, Label: label,
		TSec: sp.Samples.TSec, Split: sp.Samples.Split,
	}, nil
}
```

```typescript
// web/src/race/distance.ts
export type Samples = { tSec: number[]; split: number[] }
export type DistanceSeries = { tSec: number[]; distM: number[] }

export function reconstructDistance(s: Samples): number[] {
  const n = s.tSec.length
  if (n === 0) return []
  const out = new Array<number>(n)
  out[0] = 0
  for (let i = 1; i < n; i++) {
    const dt = s.tSec[i] - s.tSec[i - 1]
    const sp = s.split[i] || s.split[i - 1] || 0
    const v = sp > 0 ? 500 / sp : 0
    out[i] = out[i - 1] + v * dt
  }
  return out
}

// Monotone-cubic (Fritsch–Carlson) interpolation. Uses helpers from
// the live-view smoothing module if available; otherwise inline a
// minimal, distance-monotone variant here.
export function distanceAt(series: DistanceSeries, t: number): number {
  const { tSec, distM } = series
  if (tSec.length === 0) return 0
  if (t <= tSec[0]) return distM[0]
  if (t >= tSec[tSec.length - 1]) return distM[tSec.length - 1]
  let lo = 0
  let hi = tSec.length - 1
  while (hi - lo > 1) {
    const mid = (lo + hi) >> 1
    if (tSec[mid] <= t) lo = mid
    else hi = mid
  }
  return monotoneCubic(tSec, distM, lo, t)
}

function monotoneCubic(xs: number[], ys: number[], i: number, x: number): number {
  const x0 = xs[i],
    x1 = xs[i + 1],
    y0 = ys[i],
    y1 = ys[i + 1]
  const h = x1 - x0
  if (h <= 0) return y0
  const m0 = slopeAt(xs, ys, i)
  const m1 = slopeAt(xs, ys, i + 1)
  const t = (x - x0) / h
  const t2 = t * t
  const t3 = t2 * t
  const h00 = 2 * t3 - 3 * t2 + 1
  const h10 = t3 - 2 * t2 + t
  const h01 = -2 * t3 + 3 * t2
  const h11 = t3 - t2
  return h00 * y0 + h10 * h * m0 + h01 * y1 + h11 * h * m1
}

function slopeAt(xs: number[], ys: number[], i: number): number {
  const last = xs.length - 1
  if (i === 0) return (ys[1] - ys[0]) / (xs[1] - xs[0])
  if (i === last) return (ys[last] - ys[last - 1]) / (xs[last] - xs[last - 1])
  const dxL = xs[i] - xs[i - 1]
  const dxR = xs[i + 1] - xs[i]
  const slL = (ys[i] - ys[i - 1]) / dxL
  const slR = (ys[i + 1] - ys[i]) / dxR
  if (slL * slR <= 0) return 0
  return (2 * slL * slR) / (slL + slR)
}
```

```typescript
// web/src/race/useRaceOpponents.ts
import { useEffect, useRef, useState } from 'react'
import { fetchRaceOpponents, type RaceOpponentsResponse } from '../api/race'

const empty: RaceOpponentsResponse = { ghost: null, pb: null }

export function useRaceOpponents(args: {
  workoutId: string | null
  rowerId: string | null
}): RaceOpponentsResponse {
  const [state, setState] = useState<RaceOpponentsResponse>(empty)
  const lockedFor = useRef<string | null>(null)

  useEffect(() => {
    if (!args.workoutId || !args.rowerId) {
      setState(empty)
      lockedFor.current = null
      return
    }
    if (lockedFor.current === args.workoutId) return
    lockedFor.current = args.workoutId
    let cancelled = false
    fetchRaceOpponents({ workoutId: args.workoutId, rowerId: args.rowerId })
      .then((r) => {
        if (!cancelled) setState(r)
      })
      .catch(() => {
        if (!cancelled) setState(empty)
      })
    return () => {
      cancelled = true
    }
  }, [args.workoutId, args.rowerId])

  return state
}
```

```typescript
// web/src/race/RaceTile.tsx
import { BoatSprite } from './BoatSprite'
import { reconstructDistance, distanceAt, type Samples } from './distance'
import styles from './RaceTile.module.css'
import type { RaceOpponent } from '../api/race'

type Props = {
  elapsedSec: number
  userDistanceM: number
  ghost: RaceOpponent | null
  pb: RaceOpponent | null
  paneWidthPx: number
  laneHeightPx: number
}

const HULL_FLOOR = 20
const FULL_FLOOR = 28
const LANE_PX_PER_METER = 4 // tuned for tile; user always centered

export function RaceTile({
  elapsedSec,
  userDistanceM,
  ghost,
  pb,
  paneWidthPx,
  laneHeightPx,
}: Props) {
  if (laneHeightPx < HULL_FLOOR) return null
  const variant = laneHeightPx >= FULL_FLOOR ? 'full' : 'hull'

  const lanes: Array<{
    key: 'pb' | 'user' | 'ghost'
    color: string
    distanceM: number | null
    label?: string
  }> = []
  if (pb) {
    lanes.push({
      key: 'pb',
      color: '#F59E0B',
      distanceM: opponentDistanceAt(pb, elapsedSec),
      label: 'PB',
    })
  }
  lanes.push({ key: 'user', color: '#14B8A6', distanceM: userDistanceM })
  if (ghost) {
    lanes.push({
      key: 'ghost',
      color: '#94A3B8',
      distanceM: opponentDistanceAt(ghost, elapsedSec),
      label: 'Ghost',
    })
  }

  const halfWidth = paneWidthPx / 2

  return (
    <div className={styles.tile} style={{ height: lanes.length * laneHeightPx }}>
      {lanes.map((lane) => {
        const offsetM =
          lane.distanceM == null ? 0 : lane.distanceM - userDistanceM
        const offsetPx = offsetM * LANE_PX_PER_METER
        const onScreen = Math.abs(offsetPx) <= halfWidth - 12 // sprite half width
        const chip = !onScreen && lane.key !== 'user' ? formatChip(offsetM) : null
        return (
          <div
            key={lane.key}
            data-lane={lane.key}
            className={styles.lane}
            style={{ height: laneHeightPx, color: lane.color }}
          >
            {onScreen ? (
              <span
                className={styles.sprite}
                style={{ transform: `translateX(${halfWidth + offsetPx - 36}px)` }}
              >
                <BoatSprite variant={variant} />
              </span>
            ) : (
              <span
                data-testid={`chip-${lane.key}`}
                className={styles.chip}
                style={{ [offsetPx > 0 ? 'right' : 'left']: 4 } as React.CSSProperties}
              >
                {chip}
              </span>
            )}
          </div>
        )
      })}
    </div>
  )
}

function opponentDistanceAt(opp: RaceOpponent, t: number): number {
  const samples: Samples = { tSec: opp.tSec, split: opp.split }
  const distM = reconstructDistance(samples)
  if (t > opp.tSec[opp.tSec.length - 1]) {
    return distM[distM.length - 1] // ghost stops at its end
  }
  return distanceAt({ tSec: opp.tSec, distM }, t)
}

function formatChip(offsetM: number): string {
  const abs = Math.abs(offsetM)
  if (abs < 5) {
    // Convert to split-equivalent seconds at user's projected pace
    // (approximated: at 2:00/500m → 1m ≈ 0.24s; tighten in Task 11.1)
    const secs = offsetM * 0.24
    const sign = secs >= 0 ? '+' : ''
    return `${sign}${secs.toFixed(1)}s`
  }
  const sign = offsetM >= 0 ? '+' : ''
```

```typescript
// web/src/race/RaceTileAdapter.tsx
import { useEffect, useRef, useState } from 'react'
import type { LiveState } from '../live/reducer'
import { RaceTile } from './RaceTile'
import { useRaceOpponents } from './useRaceOpponents'

export function RaceTileAdapter({ liveState }: { liveState: LiveState }) {
  const ref = useRef<HTMLDivElement>(null)
  const [size, setSize] = useState({ w: 0, h: 0 })
  useEffect(() => {
    if (!ref.current) return
    const ro = new ResizeObserver((entries) => {
      const r = entries[0].contentRect
      setSize({ w: r.width, h: r.height })
    })
    ro.observe(ref.current)
    return () => ro.disconnect()
  }, [])

  const opps = useRaceOpponents({
    workoutId: liveState.workoutId ?? null,
    rowerId: liveState.rowerId ?? null,
  })

  // Lane height: divide available height by # of lanes (max 3).
  const laneCount = 1 + (opps.ghost ? 1 : 0) + (opps.pb ? 1 : 0)
  const laneHeightPx = size.h > 0 ? size.h / laneCount : 32

  return (
    <div ref={ref} style={{ width: '100%', height: '100%' }}>
      <RaceTile
        elapsedSec={liveState.elapsedSec ?? 0}
        userDistanceM={liveState.distanceM ?? 0}
        ghost={opps.ghost}
        pb={opps.pb}
        paneWidthPx={size.w}
        laneHeightPx={laneHeightPx}
      />
    </div>
  )
}
```

```typescript
import { useEffect, useState } from 'react'
import styles from './RaceResultToast.module.css'

export type RaceResult = { vsPbSec: number | null; vsGhostSec: number | null }

export function RaceResultToast({ result }: { result: RaceResult | null }) {
  const [shown, setShown] = useState<RaceResult | null>(null)
  useEffect(() => {
    if (!result) return
    setShown(result)
    const t = setTimeout(() => setShown(null), 6000)
    return () => clearTimeout(t)
  }, [result])
  if (!shown) return null
  return (
    <div className={styles.toast} role="status">
      {shown.vsPbSec != null && (
        <span>
          vs. PB:&nbsp;<strong>{format(shown.vsPbSec)}</strong>
        </span>
      )}
      {shown.vsGhostSec != null && (
        <span>
          vs. Ghost:&nbsp;<strong>{format(shown.vsGhostSec)}</strong>
        </span>
      )}
    </div>
  )
}

function format(sec: number): string {
  const sign = sec > 0 ? '+' : sec < 0 ? '' : ''
  return `${sign}${sec.toFixed(1)} s`
}
```
