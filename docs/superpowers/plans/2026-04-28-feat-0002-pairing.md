```sql
-- backend/internal/storage/migrations/0011_pairing_tokens.sql
-- FEAT-0002: per-rower pairing tokens. Each row is a single-use,
-- 24h-expiring credential allowing a fresh device to claim a rower.
-- Storage is hex(SHA-256(token)) so a leaked DB never yields plaintext.
CREATE TABLE IF NOT EXISTS pairing_tokens (
  token_hash TEXT PRIMARY KEY,
  rower_id   TEXT NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  expires_at TIMESTAMP NOT NULL,
  FOREIGN KEY (rower_id) REFERENCES rowers(id) ON DELETE CASCADE
);
CREATE INDEX IF NOT EXISTS idx_pairing_tokens_rower ON pairing_tokens(rower_id);
CREATE INDEX IF NOT EXISTS idx_pairing_tokens_expires ON pairing_tokens(expires_at);
```

```go
// backend/internal/rower/pairing.go
package rower

import (
	"context"
	"crypto/rand"
	"crypto/sha256"
	"database/sql"
	"encoding/base64"
	"encoding/hex"
	"errors"
	"time"
)

// ErrInvalidToken means the pairing token does not exist, has been
// consumed, or has expired. Callers should map to HTTP 410 Gone.
var ErrInvalidToken = errors.New("invalid pairing token")

// GenerateToken produces a 32-byte (256-bit) cryptographically random
// pairing token, base64url-encoded without padding.
func GenerateToken() (string, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(b), nil
}

// HashToken returns hex(SHA-256(plain)). Pairing tokens are 256-bit
// random secrets, so a fast hash is sufficient — Argon2id would only
// add CPU cost without security benefit (see ADR-0005 update note).
func HashToken(plain string) string {
	h := sha256.Sum256([]byte(plain))
	return hex.EncodeToString(h[:])
}

// PairingStore manages pairing_tokens rows.
type PairingStore struct{ DB *sql.DB }

func NewPairingStore(db *sql.DB) *PairingStore { return &PairingStore{DB: db} }

// Mint inserts a fresh token row and returns the plaintext token. The
// caller MUST treat the returned plaintext as the only chance to see
// it; only the hash is persisted.
func (p *PairingStore) Mint(ctx context.Context, rowerID string, ttl time.Duration) (string, error) {
	plain, err := GenerateToken()
	if err != nil {
		return "", err
	}
	expires := time.Now().Add(ttl).UTC()
	if _, err := p.DB.ExecContext(ctx,
		`INSERT INTO pairing_tokens(token_hash, rower_id, expires_at) VALUES(?, ?, ?)`,
		HashToken(plain), rowerID, expires.Format(time.RFC3339Nano)); err != nil {
		return "", err
	}
	return plain, nil
}

// Consume validates plain against pairing_tokens, deletes the row, and
// returns the bound rower_id. Single-use: a successful consume removes
// the row. Returns ErrInvalidToken when missing or expired.
func (p *PairingStore) Consume(ctx context.Context, plain string) (string, error) {
	var rowerID string
	var expiresStr string
	row := p.DB.QueryRowContext(ctx,
		`DELETE FROM pairing_tokens WHERE token_hash = ? RETURNING rower_id, expires_at`,
		HashToken(plain))
	if err := row.Scan(&rowerID, &expiresStr); err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return "", ErrInvalidToken
		}
		return "", err
	}
	exp, err := time.Parse(time.RFC3339Nano, expiresStr)
	if err != nil {
		// Stored timestamp unparseable — treat as invalid.
		return "", ErrInvalidToken
	}
	if time.Now().After(exp) {
		return "", ErrInvalidToken
	}
	return rowerID, nil
}

// PurgeExpired removes pairing_tokens rows past their expires_at and
// returns the count removed. Run from the janitor.
func (p *PairingStore) PurgeExpired(ctx context.Context) (int64, error) {
	res, err := p.DB.ExecContext(ctx,
		`DELETE FROM pairing_tokens WHERE expires_at < ?`,
		time.Now().UTC().Format(time.RFC3339Nano))
	if err != nil {
		return 0, err
	}
	return res.RowsAffected()
}
```

```go
// backend/internal/api/qr.go
package api

import (
	"github.com/skip2/go-qrcode"
)

// renderQRSVG returns an inline SVG (UTF-8 string) encoding the given
// URL at medium error correction. Size 256 is plenty for a phone
// camera at arm's length.
func renderQRSVG(url string) (string, error) {
	q, err := qrcode.New(url, qrcode.Medium)
	if err != nil {
		return "", err
	}
	// go-qrcode supports PNG natively; for SVG we use the raw bitmap +
	// inline rect rendering.
	bm := q.Bitmap() // [][]bool, true = dark
	const cell = 8
	size := len(bm) * cell
	var sb stringBuilder
	sb.WriteString(`<svg xmlns="http://www.w3.org/2000/svg" width="`)
	sb.WriteString(itoa(size))
	sb.WriteString(`" height="`)
	sb.WriteString(itoa(size))
	sb.WriteString(`" viewBox="0 0 `)
	sb.WriteString(itoa(size))
	sb.WriteString(` `)
	sb.WriteString(itoa(size))
	sb.WriteString(`" shape-rendering="crispEdges">`)
	sb.WriteString(`<rect width="100%" height="100%" fill="#fff"/>`)
	for y, row := range bm {
		for x, on := range row {
			if !on {
				continue
			}
			sb.WriteString(`<rect x="`)
			sb.WriteString(itoa(x * cell))
			sb.WriteString(`" y="`)
			sb.WriteString(itoa(y * cell))
			sb.WriteString(`" width="`)
			sb.WriteString(itoa(cell))
			sb.WriteString(`" height="`)
			sb.WriteString(itoa(cell))
			sb.WriteString(`" fill="#000"/>`)
		}
	}
	sb.WriteString(`</svg>`)
	return sb.String(), nil
}

// stringBuilder is a tiny strings.Builder wrapper to avoid the import
// dance in this single file. The actual implementation should just use
// strings.Builder; included inline only to keep the example
// self-contained.
type stringBuilder struct{ s []byte }

func (b *stringBuilder) WriteString(s string) { b.s = append(b.s, s...) }
func (b *stringBuilder) String() string       { return string(b.s) }

func itoa(n int) string {
	if n == 0 {
		return "0"
	}
	buf := [20]byte{}
	i := len(buf)
	neg := n < 0
	if neg {
		n = -n
	}
	for n > 0 {
		i--
		buf[i] = byte('0' + n%10)
		n /= 10
	}
	if neg {
		i--
		buf[i] = '-'
	}
	return string(buf[i:])
}
```

```go
// backend/internal/api/pair.go
package api

import (
	"database/sql"
	"encoding/json"
	"errors"
	"net/http"
	"time"

	"github.com/gloeckner-daniel/splits/internal/rower"
)

const pairingTokenTTL = 24 * time.Hour

// MintTokenHandler returns POST /api/v1/rowers/{id}/pairing-tokens.
// Requires the calling device to be paired with the target rower.
// Mints a single-use, 24h-expiring token; returns plaintext + URL + QR.
func MintTokenHandler(repo *rower.Repo, link *rower.DeviceLink, store *rower.PairingStore, baseURL string) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		if r.Method != http.MethodPost {
			http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
			return
		}
		id := r.PathValue("id")
		target, err := repo.Get(r.Context(), id)
		if errors.Is(err, sql.ErrNoRows) || (err == nil && target == nil) {
			http.NotFound(w, r)
			return
		}
		if err != nil {
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}
		devID := DeviceIDFromContext(r.Context())
		paired, err := link.IsPaired(r.Context(), devID, id)
		if err != nil {
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}
		if !paired {
			http.Error(w, "calling device not paired with this rower", http.StatusForbidden)
			return
		}
		plain, err := store.Mint(r.Context(), id, pairingTokenTTL)
		if err != nil {
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}
		url := baseURL + "/pair#t=" + plain
		svg, err := renderQRSVG(url)
		if err != nil {
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}
		writeJSON(w, http.StatusOK, map[string]any{
			"token":      plain,
			"url":        url,
			"qr_svg":     svg,
			"expires_at": time.Now().Add(pairingTokenTTL).UTC().Format(time.RFC3339),
		})
	}
}
```

```go
// PairExchangeHandler returns POST /api/v1/pair. Unauthenticated by
// design — the token IS the proof. On success, inserts (device_id,
// rower_id) into device_rowers and sets the active rower for the
// device. Token row is deleted (single-use).
func PairExchangeHandler(repo *rower.Repo, holder *rower.Holder, link *rower.DeviceLink, store *rower.PairingStore) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		if r.Method != http.MethodPost {
			http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
			return
		}
		var body struct {
			Token string `json:"token"`
		}
		dec := json.NewDecoder(r.Body)
		dec.DisallowUnknownFields()
		if err := dec.Decode(&body); err != nil || body.Token == "" {
			http.Error(w, "token required", http.StatusBadRequest)
			return
		}
		rowerID, err := store.Consume(r.Context(), body.Token)
		if errors.Is(err, rower.ErrInvalidToken) {
			http.Error(w, "token unknown, expired, or consumed", http.StatusGone)
			return
		}
		if err != nil {
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}
		target, err := repo.Get(r.Context(), rowerID)
		if err != nil || target == nil {
			http.Error(w, "rower vanished", http.StatusGone)
			return
		}
		devID := DeviceIDFromContext(r.Context())
		if err := link.Pair(r.Context(), devID, rowerID); err != nil {
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}
		if err := repo.SetActiveForDevice(r.Context(), devID, rowerID); err != nil {
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}
		holder.Set(target.ID, target.Name)
		w.WriteHeader(http.StatusOK)
	}
}
```

```go
func cliRevoke(ctx context.Context, db *sql.DB, rowerID string, stdout io.Writer) error {
	tx, err := db.BeginTx(ctx, nil)
	if err != nil { return err }
	defer func() { _ = tx.Rollback() }()
	res1, err := tx.ExecContext(ctx, `DELETE FROM device_rowers WHERE rower_id = ?`, rowerID)
	if err != nil { return err }
	res2, err := tx.ExecContext(ctx, `DELETE FROM pairing_tokens WHERE rower_id = ?`, rowerID)
	if err != nil { return err }
	if err := tx.Commit(); err != nil { return err }
	p, _ := res1.RowsAffected()
	t, _ := res2.RowsAffected()
	fmt.Fprintf(stdout, "revoked rower %s: %d device pairing(s), %d outstanding token(s) removed\n", rowerID, p, t)
	return nil
}
```

Add to the bottom of `0005-qr-per-rower-auth.md`:

```markdown
## Subsequent decisions (2026-04-28, FEAT-0002 v1)

- **Token hashing: SHA-256 instead of Argon2id.** Pairing tokens are
  256 bits of cryptographic randomness (`crypto/rand`), so the password-
  hardening properties of Argon2id add CPU cost without security gain.
  SHA-256(token) stored as hex; computationally indistinguishable from a
  random 256-bit string when the input is itself random.
- **No admin tier.** Trust model is flat: every paired rower can mint
  pairing URLs for new rowers / new devices. Removes `is_admin` flag
  and the "Household" admin screen from FEAT-0002 scope.
- **Token TTL: 24h, single use.** Reduces blast radius of leaked URLs
  vs. ADR's "long-lived (years)" wording. Recovery flow uses
  `splits invite` CLI to mint a fresh URL on demand.
- **No separate session cookie.** The existing `device_id` cookie
  (HttpOnly, Secure, SameSite=Lax, set by middleware) plus the
  `device_rowers` row IS the session. "Logout" = delete the row.
- **Rotation policy: cookie value never rotates.** It identifies a
```

```ts
export type PairingMint = {

export async function mintPairing(rowerID: string): Promise<PairingMint> {
```

```tsx
// web/src/pages/PairPage.tsx
import { useEffect, useState } from 'react'
import { useNavigate } from 'react-router-dom'
import { useQueryClient } from '@tanstack/react-query'
import { pair } from '../api/pairing'
import { ApiError } from '../api/error'

type Status = 'pending' | 'malformed' | 'stale' | 'error' | 'success'

function readToken(): string | null {
  const hash = window.location.hash // "#t=abc"
  if (!hash.startsWith('#t=')) return null
  const token = hash.slice(3)
  return token || null
}

export function PairPage() {
  const nav = useNavigate()
  const qc = useQueryClient()
  const [status, setStatus] = useState<Status>('pending')

  useEffect(() => {
    const tok = readToken()
    if (!tok) {
      setStatus('malformed')
      return
    }
    pair(tok)
      .then(async () => {
        // Clear the fragment from URL + history.
        window.history.replaceState(null, '', '/')
        await qc.invalidateQueries({ queryKey: ['rowers'] })
        await qc.invalidateQueries({ queryKey: ['active-rower'] })
        setStatus('success')
        nav('/', { replace: true })
      })
      .catch((e) => {
        if (e instanceof ApiError && e.status === 410) {
          setStatus('stale')
        } else {
          setStatus('error')
        }
      })
  }, [nav, qc])

  if (status === 'pending') return <p>Wird verbunden …</p>
  if (status === 'malformed') return <p>Pairing-Link ungültig.</p>
  if (status === 'stale') return <p>Code abgelaufen oder nicht mehr gültig — frag den Haushalt nach einem neuen.</p>
  if (status === 'error') return <p>Verbindung fehlgeschlagen.</p>
  return null
}
```
