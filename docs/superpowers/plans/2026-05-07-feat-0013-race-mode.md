# FEAT-0013 Race Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the FEAT-0012 race-boat tile with a first-class `race` workout mode: home-screen chip, opponent picker before start, full-takeover live view (no hero, no tiles), animated water + trails + finish banner, persisted result, dedicated post-race screen.

**Architecture:** A new `race` workout mode is added to the existing workout-mode machinery (Go `Mode` enum + SQLite CHECK constraint + frontend `FlowMode`). Race-mode workouts share `shape_hash` with `distance` workouts of the same meters, so `pb_workout`/`pb_distance` recompute is unchanged. A new endpoint `GET /api/v1/workouts/{id}/opponents` resolves opponents at session start (real ghost from `pb_workout`-style lookup, real PB from `pb_workout`/`pb_distance`, or a synth ghost generated server-side from the workout's `intensity` when no real ghost exists). The frontend detects race mode via the active workout and renders `RaceLiveView` instead of the standard hero+tiles layout. Race results are persisted as `summary_json.race`. The old `/api/v1/race/opponents` route, `RaceTile`, `RaceTileAdapter`, `RaceResultToast`, and related hooks are removed.

**Tech Stack:** Go (`net/http`, `database/sql`, SQLite migrations), React + TypeScript + Vite, vanilla CSS modules. Reuses `BoatSprite` and `web/src/race/distance.ts` from FEAT-0012; everything else under `web/src/race/` is replaced.

```go
// Package race generates synthetic opponents for the race workout
// mode (FEAT-0013). When a user starts a race for a distance they
// have never rowed before, this package builds a constant-pace ghost
// from the chosen intensity so the race is still playable.
package race

import "github.com/gloeckner-daniel/splits/internal/workout"

const sampleStepSec = 5.0

// targetSplitFor returns the seconds-per-500m for the given intensity.
// Unknown intensities map to moderat.
func targetSplitFor(in workout.Intensity) float64 {
    switch in {
    case workout.IntensityLocker:
        return 135
    case workout.IntensityHart:
        return 110
    case workout.IntensityAllOut:
        return 102
    }
    return 120
}

// SynthGhost returns parallel (t_sec, split) arrays that cover a
// constant-pace row from 0 to `meters`. Sample step is 5 s; the
// trailing sample lands on the precise total duration.
func SynthGhost(meters int, intensity workout.Intensity) ([]float64, []float64) {
    split := targetSplitFor(intensity)
    velocity := 500 / split // m/s
    totalSec := float64(meters) / velocity
    n := int(totalSec/sampleStepSec) + 1
    tSec := make([]float64, 0, n+1)
    splits := make([]float64, 0, n+1)
    for i := 0; i < n; i++ {
        tSec = append(tSec, float64(i)*sampleStepSec)
        splits = append(splits, split)
    }
    if tSec[len(tSec)-1] < totalSec {
        tSec = append(tSec, totalSec)
        splits = append(splits, split)
    }
    return tSec, splits
}
```

```go
package api

import (
    "context"
    "database/sql"
    "encoding/json"
    "errors"
    "fmt"
    "net/http"

    "github.com/gloeckner-daniel/splits/internal/race"
    "github.com/gloeckner-daniel/splits/internal/workout"
)

type RaceOpponent struct {
    Kind            string    `json:"kind"`
    SourceSessionID string    `json:"source_session_id,omitempty"`
    Label           string    `json:"label"`
    TSec            []float64 `json:"t_sec"`
    Split           []float64 `json:"split"`
}

type RaceOpponentsResponse struct {
    Ghost *RaceOpponent `json:"ghost"`
    PB    *RaceOpponent `json:"pb"`
}

// RaceOpponentsHandler serves GET /api/v1/workouts/{id}/opponents.
// Race-mode workouts only; non-race modes return 400.
func RaceOpponentsHandler(db *sql.DB) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        workoutID := r.PathValue("id")
        rowerID := r.URL.Query().Get("rower_id")
        if workoutID == "" || rowerID == "" {
            http.Error(w, "workout id and rower_id required", http.StatusBadRequest)
            return
        }
        mode, params, shape, err := loadRaceWorkout(r.Context(), db, workoutID)
        if errors.Is(err, sql.ErrNoRows) {
            http.NotFound(w, r)
            return
        }
        if err != nil {
            http.Error(w, err.Error(), http.StatusInternalServerError)
            return
        }
        if mode != string(workout.ModeRace) {
            http.Error(w, "workout is not race mode", http.StatusBadRequest)
            return
        }

        resp := RaceOpponentsResponse{}

        // Real ghost lookup: same shape_hash. Kept simple: no fallback yet.
        ghost, err := loadRealGhost(r.Context(), db, rowerID, shape)
        if err != nil {
            http.Error(w, err.Error(), http.StatusInternalServerError)
            return
        }
        if ghost == nil {
            // Synth ghost from intensity.
            var rp workout.RaceParams
            if jerr := json.Unmarshal([]byte(params), &rp); jerr != nil {
                http.Error(w, jerr.Error(), http.StatusInternalServerError)
                return
            }
            intensity, _ := loadWorkoutIntensity(r.Context(), db, workoutID)
            tSec, split := race.SynthGhost(rp.Meters, workout.Intensity(intensity))
            ghost = &RaceOpponent{Kind: "synth", Label: "ghost", TSec: tSec, Split: split}
        }
        resp.Ghost = ghost

        pb, err := loadRacePB(r.Context(), db, rowerID, shape, params)
        if err != nil {
            http.Error(w, err.Error(), http.StatusInternalServerError)
            return
        }
        resp.PB = pb

        if resp.Ghost != nil && resp.PB != nil &&
            resp.Ghost.SourceSessionID != "" &&
            resp.Ghost.SourceSessionID == resp.PB.SourceSessionID {
            resp.Ghost = nil
        }

        w.Header().Set("Content-Type", "application/json")
        _ = json.NewEncoder(w).Encode(resp)
    })
}

func loadRaceWorkout(ctx context.Context, db *sql.DB, workoutID string) (mode, params, shapeHash string, err error) {
    var sh sql.NullString
    err = db.QueryRowContext(ctx,
        `SELECT mode, params_json, COALESCE(shape_hash,'') FROM workout WHERE id = ?`, workoutID).
        Scan(&mode, &params, &sh)
    if err != nil {
        return "", "", "", err
    }
    return mode, params, sh.String, nil
}

func loadWorkoutIntensity(ctx context.Context, db *sql.DB, workoutID string) (string, error) {
    var s sql.NullString
    err := db.QueryRowContext(ctx, `SELECT intensity FROM workout WHERE id = ?`, workoutID).Scan(&s)
    if err != nil {
        return "", err
    }
    return s.String, nil
}

const realGhostQuery = `
SELECT s.id, s.summary_json
FROM sessions s
JOIN workout w ON w.id = s.workout_id
WHERE s.rower_id = ?
  AND w.shape_hash = ?
  AND s.ended_at IS NOT NULL
  AND s.deleted_at IS NULL
ORDER BY s.started_at DESC
LIMIT 1`

func loadRealGhost(ctx context.Context, db *sql.DB, rowerID, shape string) (*RaceOpponent, error) {
    if shape == "" {
        return nil, nil
    }
    var sid, summary string
    err := db.QueryRowContext(ctx, realGhostQuery, rowerID, shape).Scan(&sid, &summary)
    if errors.Is(err, sql.ErrNoRows) {
        return nil, nil

    if err != nil {
        return nil, err
    }
    tSec, split, ok := samplesFromSummary(summary)
    if !ok {
        return nil, nil
    }
    return &RaceOpponent{
        Kind: "session", SourceSessionID: sid, Label: "ghost",
        TSec: tSec, Split: split,
    }, nil
}
```

```go
// gapAtDistance returns user_total_sec - opponent_time_at_distance.
// Linear interpolation on the reconstructed (t_sec, distance) curve.
func gapAtDistance(tSec, split []float64, userDistance, userTime float64) float64 {
    if len(tSec) < 2 {
        return userTime
    }
    // Reconstruct cumulative distance at each sample.
    dist := make([]float64, len(tSec))
    for i := 1; i < len(tSec); i++ {
        dt := tSec[i] - tSec[i-1]
        v := 0.0
        if split[i] > 0 {
            v = 500.0 / split[i]
        }
        dist[i] = dist[i-1] + v*dt
    }
    if userDistance <= 0 {
        return userTime - tSec[0]
    }
    if userDistance >= dist[len(dist)-1] {
        return userTime - tSec[len(tSec)-1]
    }
    // Bracket.
    lo, hi := 0, len(dist)-1
    for hi-lo > 1 {
        mid := (lo + hi) / 2
        if dist[mid] <= userDistance {
            lo = mid
        } else {
            hi = mid
        }
    }
    span := dist[hi] - dist[lo]
    var oppTime float64
    if span <= 0 {
        oppTime = tSec[lo]
    } else {
        f := (userDistance - dist[lo]) / span
        oppTime = tSec[lo] + f*(tSec[hi]-tSec[lo])
    }
    return userTime - oppTime
}

        return nil, nil, false
    }
    if len(sp.Samples.TSec) == 0 {
        return nil, nil, false
    }
    return sp.Samples.TSec, sp.Samples.Split, true
}
```

```typescript
// web/src/api/raceOpponents.ts
import { apiFetch } from './error'

export type RaceOpponentKind = 'session' | 'synth'

export type RaceOpponent = {
  kind: RaceOpponentKind
  sourceSessionId?: string
  label: 'ghost' | 'pb'
  tSec: number[]
  split: number[]
}

export type RaceOpponentsResponse = {
  ghost: RaceOpponent | null
  pb: RaceOpponent | null
}

export async function fetchRaceOpponents(args: {
  workoutId: string
  rowerId: string
}): Promise<RaceOpponentsResponse> {
  const url = `/api/v1/workouts/${encodeURIComponent(args.workoutId)}/opponents?rower_id=${encodeURIComponent(args.rowerId)}`
  const r = await apiFetch(url)
  if (!r.ok) throw new Error(`race opponents fetch failed: ${r.status}`)
  type Wire = {
    ghost: { kind: 'session' | 'synth'; source_session_id?: string; label: 'ghost' | 'pb'; t_sec: number[]; split: number[] } | null
    pb: { kind: 'session' | 'synth'; source_session_id?: string; label: 'ghost' | 'pb'; t_sec: number[]; split: number[] } | null
  }
  const w = (await r.json()) as Wire
  const map = (o: Wire['ghost']): RaceOpponent | null => o ? {
    kind: o.kind,
    sourceSessionId: o.source_session_id,
    label: o.label,
    tSec: o.t_sec ?? [],
    split: o.split ?? [],
  } : null
  return { ghost: map(w.ghost), pb: map(w.pb) }
}
```

```typescript
// useRaceData.ts
import { useEffect, useState } from 'react'
import { fetchRaceOpponents, type RaceOpponentsResponse } from '../api/raceOpponents'

const empty: RaceOpponentsResponse = { ghost: null, pb: null }

function readSelection(): { ghost: boolean; pb: boolean } {
  try {
    const raw = sessionStorage.getItem('race-opp-selection')
    if (!raw) return { ghost: true, pb: true }
    return JSON.parse(raw) as { ghost: boolean; pb: boolean }
  } catch {
    return { ghost: true, pb: true }
  }
}

export function useRaceData(args: { workoutId: string | null; rowerId: string | null }): RaceOpponentsResponse {
  const [state, setState] = useState<RaceOpponentsResponse>(empty)

  useEffect(() => {
    if (!args.workoutId || !args.rowerId) {
      setState(empty)
      return
    }
    let cancelled = false
    fetchRaceOpponents({ workoutId: args.workoutId, rowerId: args.rowerId })
      .then((r) => {
        if (cancelled) return
        const sel = readSelection()
        setState({
          ghost: sel.ghost ? r.ghost : null,
          pb: sel.pb ? r.pb : null,
        })
      })
      .catch(() => { if (!cancelled) setState(empty) })
    return () => { cancelled = true }
  }, [args.workoutId, args.rowerId])

  return state
}
```

```typescript
// reducedMotion.ts
import { useEffect, useState } from 'react'

export function useReducedMotion(): boolean {
  const [r, setR] = useState(() => {
    if (typeof window === 'undefined' || !window.matchMedia) return false
    return window.matchMedia('(prefers-reduced-motion: reduce)').matches
  })
  useEffect(() => {
    if (typeof window === 'undefined' || !window.matchMedia) return
    const mq = window.matchMedia('(prefers-reduced-motion: reduce)')
    const onChange = () => setR(mq.matches)
    mq.addEventListener?.('change', onChange)
    return () => mq.removeEventListener?.('change', onChange)
  }, [])
  return r
}
```

```tsx
// StatBar.tsx
import styles from './StatBar.module.css'

type Variant = 'phone' | 'tablet'

type Props = {
  variant: Variant
  splitSec: number
  distanceLeftM: number
  watts: number
  spm?: number
  vsPbSec: number | null
  vsGhostSec: number | null
}

function fmtSplit(s: number): string {
  const m = Math.floor(s / 60)
  const r = Math.round(s - m * 60)
  return `${m}:${String(r).padStart(2, '0')}`
}
function fmtGap(n: number): string {
  const sign = n > 0 ? '+' : n < 0 ? '' : ''
  return `${sign}${n.toFixed(1)}s`
}

function leaderGap(pb: number | null, ghost: number | null): { label: string; value: number } | null {
  // Leader = the opponent ahead of user (positive gap means user is behind).
  const cands = [
    pb != null ? { label: 'PB', value: pb } : null,
    ghost != null ? { label: 'Ghost', value: ghost } : null,
  ].filter((x): x is { label: string; value: number } => x != null)
  if (cands.length === 0) return null
  const ahead = cands.filter((c) => c.value > 0)
  if (ahead.length > 0) {
    ahead.sort((a, b) => b.value - a.value)
    return ahead[0]
  }
  // No opponent ahead — show user's lead over closest opponent (least negative).
  cands.sort((a, b) => b.value - a.value)
  return cands[0]
}

export function StatBar(p: Props) {
  if (p.variant === 'phone') {
    const lead = leaderGap(p.vsPbSec, p.vsGhostSec)
    return (
      <div className={styles.bar} data-variant="phone">
        <span className={styles.cell}>{fmtSplit(p.splitSec)}</span>
        <span className={styles.cell}>{p.distanceLeftM} m</span>
        <span className={styles.cell}>{p.watts} W</span>
        {lead && <span className={styles.cell}>{lead.label} {fmtGap(lead.value)}</span>}
      </div>
    )
  }
  return (
    <div className={styles.bar} data-variant="tablet">
      <span className={styles.cell}>{fmtSplit(p.splitSec)}</span>
      <span className={styles.cell}>{p.distanceLeftM} m</span>
      <span className={styles.cell}>{p.watts} W</span>
      <span className={styles.cell}>{p.spm ?? 0}</span>
      {p.vsPbSec != null && <span className={styles.cell}>PB {fmtGap(p.vsPbSec)}</span>}
      {p.vsGhostSec != null && <span className={styles.cell}>Ghost {fmtGap(p.vsGhostSec)}</span>}
    </div>
  )
}
```

```tsx
// RaceCanvas.tsx
import { useMemo } from 'react'
import type { RaceOpponent } from '../api/raceOpponents'
import { reconstructDistance, distanceAt } from './distance'
import { BoatSprite } from './BoatSprite'
import styles from './RaceCanvas.module.css'

type Props = {
  elapsedSec: number
  userDistanceM: number
  totalMeters: number
  ghost: RaceOpponent | null
  pb: RaceOpponent | null
  viewportWidth: number
  viewportHeight: number
  reducedMotion: boolean
}

const LANE_PX_PER_METER = 4
const SPRITE_HALF_WIDTH = 12
const TABLET_WIDTH = 768
const TRAIL_OFFSETS_SEC = [-5, -10, -15]
const TRAIL_OPACITIES = [0.35, 0.20, 0.10]

export function RaceCanvas({
  elapsedSec, userDistanceM, totalMeters, ghost, pb,
  viewportWidth, viewportHeight, reducedMotion,
}: Props) {
  const lanes = useMemo(() => {
    const out: Array<{ kind: 'pb' | 'user' | 'ghost'; opp: RaceOpponent | null; color: string }> = []
    if (pb) out.push({ kind: 'pb', opp: pb, color: '#F59E0B' })
    out.push({ kind: 'user', opp: null, color: '#14B8A6' })
    if (ghost) out.push({ kind: 'ghost', opp: ghost, color: '#94A3B8' })
    return out
  }, [pb, ghost])

  const laneHeight = viewportHeight / lanes.length
  const centerX = viewportWidth / 2

  // Distance markers
  const isTablet = viewportWidth >= TABLET_WIDTH
  const markerStep = isTablet ? 100 : 250
  const markersFit = (totalMeters / markerStep) >= 6
  const markers = isTablet && markersFit
    ? Array.from({ length: Math.floor(totalMeters / markerStep) }, (_, i) => (i + 1) * markerStep)
    : []

  // Banner
  const showBanner = viewportWidth >= TABLET_WIDTH

  // Finish line position relative to user
  const finishOffsetPx = (totalMeters - userDistanceM) * LANE_PX_PER_METER

  return (
    <div className={styles.canvas} style={{ width: viewportWidth, height: viewportHeight }}>
      {showBanner && <div className={styles.banner}>{totalMeters} m Rennen</div>}
      <div
        className={styles.water}
        data-water-animated={reducedMotion ? undefined : 'true'}
      />
      {lanes.map((lane) => (
        <Lane
          key={lane.kind}
          kind={lane.kind}
          opp={lane.opp}
          color={lane.color}
          height={laneHeight}
          centerX={centerX}
          paneWidth={viewportWidth}
          elapsedSec={elapsedSec}
          userDistanceM={userDistanceM}
        />
      ))}
      {/* Distance markers */}
      {markers.map((m) => {
        const offsetPx = (m - userDistanceM) * LANE_PX_PER_METER
        const x = centerX + offsetPx
        if (x < 0 || x > viewportWidth) return null
        return <div key={m} data-marker className={styles.marker} style={{ left: x }}>·{m}·</div>
      })}
      {/* Finish line + flag */}
      {finishOffsetPx >= -SPRITE_HALF_WIDTH && finishOffsetPx <= viewportWidth - centerX && (
        <>
          <div className={styles.finishLine} style={{ left: centerX + finishOffsetPx }} />
          {isTablet && (
            <div data-finish="flag" className={styles.finishFlag} style={{ left: centerX + finishOffsetPx }}>🏁</div>
          )}
        </>
      )}
    </div>
  )
}

function Lane({ kind, opp, color, height, centerX, paneWidth, elapsedSec, userDistanceM }: {
  kind: 'pb' | 'user' | 'ghost'
  opp: RaceOpponent | null
  color: string
  height: number
  centerX: number
  paneWidth: number
  elapsedSec: number
  userDistanceM: number
}) {
  const series = useMemo(() => {
    if (!opp) return null
    const distM = reconstructDistance({ tSec: opp.tSec, split: opp.split })
    return { tSec: opp.tSec, distM }
  }, [opp])

  const oppDistanceM = series == null
    ? userDistanceM
    : (elapsedSec > series.tSec[series.tSec.length - 1]
        ? series.distM[series.distM.length - 1]
        : distanceAt(series, elapsedSec))

  const offsetMeters = oppDistanceM - userDistanceM
  const offsetPx = offsetMeters * LANE_PX_PER_METER
  const onScreen = Math.abs(offsetPx) <= paneWidth / 2 - SPRITE_HALF_WIDTH

  return (
    <div className={styles.lane} data-lane={kind} style={{ height, color }}>
      {/* Trails (opponents only) */}
      {series && [0, 1, 2].map((i) => {
        const lookback = TRAIL_OFFSETS_SEC[i]
        const t = elapsedSec + lookback
        if (t < 0) return null
        const d = t > series.tSec[series.tSec.length - 1]
          ? series.distM[series.distM.length - 1]
          : distanceAt(series, t)
        const tx = centerX + (d - userDistanceM) * LANE_PX_PER_METER - SPRITE_HALF_WIDTH
        return (
          <span
            key={i}
            data-trail
            className={styles.sprite}
            style={{ left: tx, opacity: TRAIL_OPACITIES[i] }}
          >
            <BoatSprite variant="hull" />
          </span>
        )
      })}
      {/* Main sprite */}
      {onScreen && (
        <span
          className={styles.sprite}
          style={{ left: centerX + offsetPx - SPRITE_HALF_WIDTH }}
        >
          <BoatSprite variant="full" />
        </span>
      )}
      {!onScreen && opp && (
        <span
          data-testid={`chip-${kind}`}
          className={styles.chip}
          style={{ [offsetPx > 0 ? 'right' : 'left']: 4 } as React.CSSProperties}
        >
          {formatChip(offsetMeters)}
        </span>
      )}
    </div>
  )
}

function formatChip(offsetMeters: number): string {
  const abs = Math.abs(offsetMeters)
  if (abs < 5) {
    const sec = offsetMeters * 0.24
    return `${sec >= 0 ? '+' : ''}${sec.toFixed(1)}s`
  }
  return `${offsetMeters >= 0 ? '+' : ''}${Math.round(offsetMeters)}m`
}
```

```tsx
import type { LiveState } from '../live/reducer'
import { StatBar } from './StatBar'

import styles from './RaceLiveView.module.css'

type Props = { liveState: LiveState; totalMeters: number }

export function RaceLiveView({ liveState, totalMeters }: Props) {
  const ref = useRef<HTMLDivElement>(null)
  const [size, setSize] = useState({ w: 0, h: 0 })
  useEffect(() => {
    if (!ref.current) return
    const ro = new ResizeObserver((entries) => {
      const r = entries[0].contentRect
      setSize({ w: r.width, h: r.height })
    })
    ro.observe(ref.current)
    return () => ro.disconnect()
  }, [])
  const reducedMotion = useReducedMotion()
  const opps = useRaceData({

    rowerId: liveState.session?.rower_id ?? null,
  })

  const elapsedSec = liveState.stroke?.time_sec ?? 0
  const userDistanceM = liveState.stroke?.distance_m ?? 0
  const split = liveState.stroke?.pace_sec_per_500m ?? 120
  const watts = liveState.stroke?.power_w ?? 0
  const spm = liveState.stroke?.spm ?? 0

  // Compute gaps for stat bar
  const ghostGap = opps.ghost ? gapAtTime(opps.ghost.tSec, opps.ghost.split, userDistanceM, elapsedSec) : null
  const pbGap = opps.pb ? gapAtTime(opps.pb.tSec, opps.pb.split, userDistanceM, elapsedSec) : null

  const variant = size.w >= 768 ? 'tablet' : 'phone'
  const canvasH = Math.max(0, size.h - 56) // 56px stat bar

  return (
    <div ref={ref} className={styles.view}>
      <div data-component="stat-bar"><StatBar
        variant={variant}
        splitSec={split}
        distanceLeftM={Math.max(0, totalMeters - userDistanceM)}
        watts={watts}
        spm={spm}
        vsPbSec={pbGap}
        vsGhostSec={ghostGap}
      /></div>
      <div data-component="race-canvas"><RaceCanvas
        elapsedSec={elapsedSec}
        userDistanceM={userDistanceM}
        totalMeters={totalMeters}
        ghost={opps.ghost}
        pb={opps.pb}
        viewportWidth={size.w}
        viewportHeight={canvasH}
        reducedMotion={reducedMotion}
      /></div>
    </div>
  )
}

// Linear-interp inverse: opponent time at user's distance, then user_time - opp_time.
function gapAtTime(tSec: number[], split: number[], userDist: number, userTime: number): number {
  if (tSec.length < 2) return 0
  const distM: number[] = [0]
  for (let i = 1; i < tSec.length; i++) {
    const dt = tSec[i] - tSec[i - 1]
    const v = split[i] > 0 ? 500 / split[i] : 0
    distM.push(distM[i - 1] + v * dt)
  }
  if (userDist <= 0) return userTime - tSec[0]
  if (userDist >= distM[distM.length - 1]) return userTime - tSec[tSec.length - 1]
  let lo = 0, hi = distM.length - 1
  while (hi - lo > 1) {
    const mid = (lo + hi) >> 1
    if (distM[mid] <= userDist) lo = mid; else hi = mid
  }
  const span = distM[hi] - distM[lo]
  const oppTime = span <= 0 ? tSec[lo] : tSec[lo] + (userDist - distM[lo]) / span * (tSec[hi] - tSec[lo])
  return userTime - oppTime
}
```

```tsx
// PaceOverlayChart.tsx
import { reconstructDistance } from './distance'

type Series = { color: string; label: string; tSec: number[]; split: number[] }
type Props = { series: Series[]; totalMeters: number; width: number; height: number }

export function PaceOverlayChart({ series, totalMeters, width, height }: Props) {
  const padding = { left: 32, right: 8, top: 8, bottom: 24 }
  const xRange = [padding.left, width - padding.right]
  const yRange = [padding.top, height - padding.bottom]

  const all = series.map((s) => ({
    color: s.color,
    label: s.label,
    points: zipDistanceSplit(s),
  }))
  const allSplits = all.flatMap((s) => s.points.map((p) => p.split))
  const yMin = Math.min(...allSplits) * 0.95
  const yMax = Math.max(...allSplits) * 1.05

  const xAt = (m: number) => xRange[0] + (m / totalMeters) * (xRange[1] - xRange[0])
  const yAt = (s: number) => yRange[1] - ((s - yMin) / (yMax - yMin)) * (yRange[1] - yRange[0])

  return (
    <svg width={width} height={height} role="img" aria-label="Pace overlay">
      <line x1={xRange[0]} y1={yRange[1]} x2={xRange[1]} y2={yRange[1]} stroke="currentColor" strokeOpacity="0.3" />
      <line x1={xRange[0]} y1={yRange[0]} x2={xRange[0]} y2={yRange[1]} stroke="currentColor" strokeOpacity="0.3" />
      {all.map((s, i) => (
        <polyline
          key={i}
          fill="none"
          stroke={s.color}
          strokeWidth={2}
          points={s.points.map((p) => `${xAt(p.distance)},${yAt(p.split)}`).join(' ')}
        />
      ))}
    </svg>
  )
}

function zipDistanceSplit(s: Series): { distance: number; split: number }[] {
  const distM = reconstructDistance({ tSec: s.tSec, split: s.split })
  return distM.map((d, i) => ({ distance: d, split: s.split[i] }))
}
```

```tsx
// RacePostScreen.tsx
import { PaceOverlayChart } from './PaceOverlayChart'
import type { RaceOpponent } from '../api/raceOpponents'
import styles from './RacePostScreen.module.css'

type Props = {
  totalMeters: number
  userTotalSec: number
  userTSec: number[]
  userSplit: number[]
  ghost: RaceOpponent | null
  pb: RaceOpponent | null
  vsGhostSec: number | null
  vsPbSec: number | null
  onRetry: () => void
  onShowDetail: () => void
}

function fmtMmSs(s: number): string {
  const m = Math.floor(s / 60)
  const r = Math.round(s - m * 60)
  return `${m}:${String(r).padStart(2, '0')}`
}

export function RacePostScreen(p: Props) {
  const series = [
    { color: '#14B8A6', label: 'Du', tSec: p.userTSec, split: p.userSplit },
    ...(p.pb ? [{ color: '#F59E0B', label: 'PB', tSec: p.pb.tSec, split: p.pb.split }] : []),
    ...(p.ghost ? [{ color: '#94A3B8', label: 'Ghost', tSec: p.ghost.tSec, split: p.ghost.split }] : []),
  ]
  return (
    <div className={styles.screen}>
      <h1 className={styles.title}>{p.totalMeters} m Rennen</h1>
      <div className={styles.summary}>
        <div className={styles.cell}>
          <span className={styles.label}>Endzeit</span>
          <span className={styles.value}>{fmtMmSs(p.userTotalSec)}</span>
        </div>
        {p.vsPbSec != null && (
          <div className={styles.cell}>
            <span className={styles.label}>vs. PB</span>
            <span className={styles.value}>{p.vsPbSec >= 0 ? '+' : ''}{p.vsPbSec.toFixed(1)} s</span>
          </div>
        )}
        {p.vsGhostSec != null && (
          <div className={styles.cell}>
            <span className={styles.label}>vs. Ghost</span>
            <span className={styles.value}>{p.vsGhostSec >= 0 ? '+' : ''}{p.vsGhostSec.toFixed(1)} s</span>
          </div>
        )}
      </div>
      <div className={styles.chart}>
        <PaceOverlayChart series={series} totalMeters={p.totalMeters} width={600} height={240} />
      </div>
      <div className={styles.actions}>
        <button onClick={p.onRetry}>Erneut rennen</button>
        <button onClick={p.onShowDetail}>Detail</button>
      </div>
    </div>
  )
}
```
